Agrega en el <script setup> la función calcularHoraFin y los campos en el form:

    // En useForm, agrega/reemplaza los campos de hora:
    const form = useForm({
        // ... campos existentes (cliente_id, restaurantero_id, fecha, notas) ...
        hora_inicio: '',
        duracion:    30,  // default 30 minutos
    });

    // Función para calcular hora de fin
    const calcularHoraFin = (horaInicio, duracionMinutos) => {
        if (!horaInicio || !duracionMinutos) return '';
        const [h, m] = horaInicio.split(':').map(Number);
        const totalMinutos = h * 60 + m + Number(duracionMinutos);
        const hFin = Math.floor(totalMinutos / 60) % 24;
        const mFin = totalMinutos % 60;
        return `${String(hFin).padStart(2, '0')}:${String(mFin).padStart(2, '0')}`;
    };
