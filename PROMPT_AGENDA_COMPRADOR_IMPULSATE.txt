FLUJO ANTERIOR (ya NO aplica):
  Admin selecciona Proveedor → asigna Compradores a slots → envía al Proveedor

FLUJO NUEVO (este prompt):
  Admin selecciona Comprador → asigna Proveedores a slots → envía al Comprador
  Comprador acepta  → se crean las citas automáticamente en el sistema
  Comprador rechaza → el admin recibe notificación para hacer una re-propuesta

REGLA CLAVE: Solo participan en la agenda los usuarios APROBADOS en el evento activo.
  - Compradores disponibles: evento_usuario WHERE tipo='comprador' AND estado='aprobado'
  - Proveedores disponibles: evento_usuario WHERE tipo='proveedor' AND estado='aprobado'

    <?php
    namespace App\Models;

    use Illuminate\Database\Eloquent\Model;
    use Illuminate\Database\Eloquent\Relations\BelongsTo;
    use Illuminate\Database\Eloquent\Relations\HasMany;
    use Illuminate\Support\Str;

    class AgendaPropuesta extends Model
    {
        protected $fillable = [
            'evento_id', 'user_id', 'admin_id',
            'estado', 'token', 'enviada_at', 'respondida_at',
        ];

        protected $casts = [
            'enviada_at'    => 'datetime',
            'respondida_at' => 'datetime',
        ];

        // El COMPRADOR que recibe la propuesta
        public function comprador(): BelongsTo
        {
            return $this->belongsTo(User::class, 'user_id');
        }

        public function evento(): BelongsTo
        {
            return $this->belongsTo(Evento::class);
        }

        public function admin(): BelongsTo
        {
            return $this->belongsTo(User::class, 'admin_id');
        }

        // Las citas propuestas (proveedor + slot)
        public function citas(): HasMany
        {
            return $this->hasMany(AgendaPropuestaCita::class);
        }

        public function generarToken(): string
        {
            $this->token = Str::random(64);
            $this->save();
            return $this->token;
        }

        public function esPendiente(): bool
        {
            return $this->estado === 'pendiente';
        }
    }

    <?php
    namespace App\Http\Controllers\Admin;

    use App\Http\Controllers\Controller;
    use App\Models\AgendaPropuesta;
    use App\Models\AgendaPropuestaCita;
    use App\Models\Evento;
    use App\Models\Restaurantero;
    use App\Models\User;
    use Illuminate\Http\Request;
    use Illuminate\Support\Facades\DB;
    use Illuminate\Support\Facades\Mail;
    use Inertia\Inertia;

    class AgendaController extends Controller
    {
        // ── LISTA DE PROPUESTAS ──────────────────────────────────────────────
        public function index()
        {
            $propuestas = AgendaPropuesta::with([
                'comprador:id,name,nombre_empresa,email',
                'citas.proveedor:id,nombre,foto',
                'evento:id,nombre',
            ])
            ->latest()
            ->paginate(20);

            return Inertia::render('Admin/Agenda/Index', [
                'propuestas' => $propuestas,
            ]);
        }

        // ── FORMULARIO CREAR ─────────────────────────────────────────────────
        public function crear()
        {
            $evento = Evento::where('activo', true)->latest()->first();

            if (!$evento) {
                return redirect()->route('admin.agenda.index')
                    ->with('error', 'No hay un evento activo.');
            }

            // COMPRADORES aprobados en el evento
            $compradores = User::whereIn('id', function ($q) use ($evento) {
                $q->select('user_id')
                  ->from('evento_usuario')
                  ->where('evento_id', $evento->id)
                  ->where('tipo',      'comprador')
                  ->where('estado',    'aprobado');
            })
            ->select('id', 'name', 'nombre_empresa', 'municipio', 'camara_asociacion')
            ->orderBy('nombre_empresa')->orderBy('name')
            ->get()
            ->map(fn($u) => [
                'id'       => $u->id,
                'nombre'   => $u->nombre_empresa ?? $u->name,
                'municipio'=> $u->municipio ?? '',
                'canirac'  => $u->camara_asociacion === 'CANIRAC',
            ]);

            // PROVEEDORES aprobados en el evento
            $proveedores = Restaurantero::whereIn('user_id', function ($q) use ($evento) {
                $q->select('user_id')
                  ->from('evento_usuario')
                  ->where('evento_id', $evento->id)
                  ->where('tipo',      'proveedor')
                  ->where('estado',    'aprobado');
            })
            ->select('id', 'nombre', 'foto', 'descripcion', 'categoria_id')
            ->with('categoria:id,nombre')
            ->orderBy('nombre')
            ->get()
            ->map(fn($r) => [
                'id'        => $r->id,
                'nombre'    => $r->nombre,
                'foto'      => $r->foto,
                'categoria' => $r->categoria->nombre ?? ($r->categoria ?? ''),
            ]);

            // Slots del día del evento (09:00–17:00, cada 30 min)
            $fechaEvento = $evento->fecha_inicio
                ? \Carbon\Carbon::parse($evento->fecha_inicio)->format('Y-m-d')
                : '2026-07-21';

            $slots = [];
            $cursor = \Carbon\Carbon::parse($fechaEvento . ' 09:00:00');
            $limite = \Carbon\Carbon::parse($fechaEvento . ' 17:00:00');
            while ($cursor->lt($limite)) {
                $slots[] = [
                    'inicio' => $cursor->toIso8601String(),
                    'fin'    => $cursor->copy()->addMinutes(30)->toIso8601String(),
                    'label'  => $cursor->format('H:i'),
                ];
                $cursor->addMinutes(30);
            }

            return Inertia::render('Admin/Agenda/Crear', [
                'evento'      => $evento,
                'compradores' => $compradores,
                'proveedores' => $proveedores,
                'slots'       => $slots,
            ]);
        }

        // ── GUARDAR PROPUESTA ────────────────────────────────────────────────
        public function store(Request $request)
        {
            $request->validate([
                'user_id'                    => 'required|exists:users,id',
                'citas'                      => 'required|array|min:1',
                'citas.*.restaurantero_id'   => 'required|exists:restauranteros,id',
                'citas.*.slot_inicio'        => 'required|date',
                'citas.*.slot_fin'           => 'required|date|after:citas.*.slot_inicio',
                'enviar'                     => 'boolean',
            ]);

            $evento = Evento::where('activo', true)->latest()->firstOrFail();

            // Verificar que no exista propuesta pendiente para este comprador
            $existente = AgendaPropuesta::where('evento_id', $evento->id)
                ->where('user_id', $request->user_id)
                ->where('estado', 'pendiente')
                ->exists();

            if ($existente) {
                return back()->withErrors([
                    'error' => 'Este comprador ya tiene una propuesta pendiente de respuesta.'
                ]);
            }

            // Verificar que no se repita un proveedor en dos slots de la misma propuesta
            $restauranteroIds = array_column($request->citas, 'restaurantero_id');
            if (count($restauranteroIds) !== count(array_unique($restauranteroIds))) {
                return back()->withErrors([
                    'error' => 'No puedes asignar el mismo proveedor dos veces en la misma propuesta.'
                ]);
            }

            $propuesta = null;

            DB::transaction(function () use ($request, $evento, &$propuesta) {
                $propuesta = AgendaPropuesta::create([
                    'evento_id' => $evento->id,
                    'user_id'   => $request->user_id,
                    'admin_id'  => auth()->id(),
                    'estado'    => 'pendiente',
                ]);

                foreach ($request->citas as $cita) {
                    AgendaPropuestaCita::create([
                        'agenda_propuesta_id' => $propuesta->id,
                        'restaurantero_id'    => $cita['restaurantero_id'],
                        'slot_inicio'         => $cita['slot_inicio'],
                        'slot_fin'            => $cita['slot_fin'],
                    ]);
                }

                if ($request->boolean('enviar')) {
                    $this->enviarCorreo($propuesta->load([
                        'comprador', 'citas.proveedor', 'evento'
                    ]));
                }
            });

            return redirect()->route('admin.agenda.index')
                ->with('success', 'Propuesta ' . ($request->boolean('enviar') ? 'enviada al comprador.' : 'guardada como borrador.'));
        }

        // ── ENVIAR CORREO MANUALMENTE ────────────────────────────────────────
        public function enviar(AgendaPropuesta $agenda)
        {
            if (!$agenda->esPendiente()) {
                return back()->withErrors(['error' => 'Solo se pueden enviar propuestas pendientes.']);
            }

            $this->enviarCorreo($agenda->load(['comprador', 'citas.proveedor', 'evento']));

            return back()->with('success', 'Correo enviado al comprador.');
        }

        // ── ELIMINAR ─────────────────────────────────────────────────────────
        public function destroy(AgendaPropuesta $agenda)
        {
            if ($agenda->estado === 'aceptada') {
                return back()->withErrors(['error' => 'No puedes eliminar una propuesta ya aceptada.']);
            }
            $agenda->delete();
            return back()->with('success', 'Propuesta eliminada.');
        }

        // ── MÉTODO PRIVADO: ENVÍO DE CORREO ─────────────────────────────────
        private function enviarCorreo(AgendaPropuesta $propuesta): void
        {
            $token      = $propuesta->generarToken();
            $urlAceptar = route('agenda.aceptar', $token);
            $urlRechazar= route('agenda.rechazar', $token);

            // Construir lista de citas para el cuerpo del correo
            $listaCitas = $propuesta->citas->sortBy('slot_inicio')->map(function ($c) {
                $hora = \Carbon\Carbon::parse($c->slot_inicio)->format('H:i') . ' – ' .
                        \Carbon\Carbon::parse($c->slot_fin)->format('H:i');
                return $hora . ' → ' . ($c->proveedor->nombre ?? 'Proveedor');
            })->join("\n");

            $fechaEvento = $propuesta->evento->fecha_inicio
                ? \Carbon\Carbon::parse($propuesta->evento->fecha_inicio)
                    ->locale('es')->isoFormat('D [de] MMMM [de] YYYY')
                : '21 de julio de 2026';

            $nombreComprador = $propuesta->comprador->nombre_empresa
                               ?? $propuesta->comprador->name;

            // Usa PlantillaCorreo si existe, o Mailable directo
            $plantilla = \App\Models\PlantillaCorreo::where('clave', 'agenda_propuesta_comprador')
                ->where('activo', true)->first();

            if ($plantilla) {
                $mail = new \App\Mail\PlantillaCorreoMail($plantilla, [
                    '{{nombre_comprador}}' => $nombreComprador,
                    '{{nombre_evento}}'    => $propuesta->evento->nombre,
                    '{{fecha_evento}}'     => $fechaEvento,
                    '{{lista_citas}}'      => $listaCitas,
                    '{{url_aceptar}}'      => $urlAceptar,
                    '{{url_rechazar}}'     => $urlRechazar,
                ]);
                Mail::to($propuesta->comprador->email)->send($mail);
            } else {
                // Fallback: Mailable directo (ver paso 5)
                Mail::to($propuesta->comprador->email)
                    ->send(new \App\Mail\AgendaPropuestaCompradorMail(
                        $propuesta, $urlAceptar, $urlRechazar
                    ));
            }

            $propuesta->update(['enviada_at' => now()]);
        }
    }

    <?php
    namespace App\Http\Controllers;

    use App\Models\AgendaPropuesta;
    use App\Models\Cita;
    use Illuminate\Support\Facades\DB;
    use Inertia\Inertia;

    class AgendaPublicaController extends Controller
    {
        // Muestra la propuesta al comprador (página pública, sin auth)
        public function ver(string $token)
        {
            $propuesta = AgendaPropuesta::where('token', $token)
                ->with([
                    'comprador:id,name,nombre_empresa',
                    'citas.proveedor:id,nombre,foto,descripcion',
                    'evento:id,nombre,fecha_inicio',
                ])
                ->firstOrFail();

            return Inertia::render('Agenda/Responder', [
                'propuesta' => $propuesta,
                'token'     => $token,
                'yaRespondida' => !$propuesta->esPendiente(),
            ]);
        }

        // El COMPRADOR acepta la propuesta
        public function aceptar(string $token)
        {
            $propuesta = AgendaPropuesta::where('token', $token)
                ->where('estado', 'pendiente')
                ->with(['citas.proveedor', 'comprador', 'evento'])
                ->firstOrFail();

            DB::transaction(function () use ($propuesta) {
                $propuesta->update([
                    'estado'        => 'aceptada',
                    'respondida_at' => now(),
                ]);

                // Crear una Cita por cada proveedor asignado
                foreach ($propuesta->citas as $citaPropuesta) {
                    // Lee tu modelo Cita para saber los campos exactos.
                    // Los campos comunes son:
                    Cita::create([
                        'evento_id'        => $propuesta->evento_id,
                        'restaurantero_id' => $citaPropuesta->restaurantero_id,
                        'cliente_id'       => $propuesta->user_id,
                        'inicio'           => $citaPropuesta->slot_inicio,
                        'fin'              => $citaPropuesta->slot_fin,
                        'estado'           => 'confirmada',
                        // Agrega aquí cualquier otro campo requerido en tu modelo Cita
                    ]);

                    // Notificar al proveedor que tiene una nueva cita confirmada
                    $proveedor = $citaPropuesta->proveedor;
                    if ($proveedor && class_exists(\App\Models\Notificacion::class)) {
                        \App\Models\Notificacion::crear(
                            $proveedor->user_id,
                            'cita_confirmada',
                            '¡Nueva cita confirmada!',
                            'El comprador ' .
                            ($propuesta->comprador->nombre_empresa ?? $propuesta->comprador->name) .
                            ' confirmó su cita contigo el ' .
                            \Carbon\Carbon::parse($citaPropuesta->slot_inicio)
                                ->format('d/m/Y H:i') . '.'
                        );
                    }
                }

                // Notificar al comprador
                if (class_exists(\App\Models\Notificacion::class)) {
                    \App\Models\Notificacion::crear(
                        $propuesta->user_id,
                        'agenda_aceptada',
                        '¡Agenda confirmada!',
                        'Has aceptado tu agenda de citas para el evento "' .
                        $propuesta->evento->nombre . '". Tus citas están confirmadas.'
                    );
                }
            });

            return Inertia::render('Agenda/Gracias', [
                'tipo'      => 'aceptada',
                'propuesta' => $propuesta->load(['citas.proveedor', 'evento']),
            ]);
        }

        // El COMPRADOR rechaza la propuesta
        public function rechazar(string $token)
        {
            $propuesta = AgendaPropuesta::where('token', $token)
                ->where('estado', 'pendiente')
                ->with(['comprador', 'evento'])
                ->firstOrFail();

            $propuesta->update([
                'estado'        => 'rechazada',
                'respondida_at' => now(),
            ]);

            // Notificar a TODOS los admins sobre el rechazo para re-propuesta
            $admins = \App\Models\User::role(['admin', 'super-admin'])->get();
            $nombreComprador = $propuesta->comprador->nombre_empresa
                               ?? $propuesta->comprador->name;

            foreach ($admins as $admin) {
                if (class_exists(\App\Models\Notificacion::class)) {
                    \App\Models\Notificacion::crear(
                        $admin->id,
                        'agenda_rechazada',
                        'Propuesta de agenda rechazada',
                        'El comprador "' . $nombreComprador . '" rechazó su propuesta de agenda para el evento "' .
                        $propuesta->evento->nombre . '". Es necesario hacer una re-propuesta.'
                    );
                }
            }

            // Notificar al comprador también
            if (class_exists(\App\Models\Notificacion::class)) {
                \App\Models\Notificacion::crear(
                    $propuesta->user_id,
                    'agenda_rechazada',
                    'Propuesta rechazada',
                    'Has rechazado la propuesta de agenda. El equipo de IMPULSATE te contactará pronto.'
                );
            }

            return Inertia::render('Agenda/Gracias', [
                'tipo'      => 'rechazada',
                'propuesta' => $propuesta,
            ]);
        }
    }

    <?php
    namespace App\Mail;

    use App\Models\AgendaPropuesta;
    use Illuminate\Bus\Queueable;
    use Illuminate\Mail\Mailable;
    use Illuminate\Queue\SerializesModels;

    class AgendaPropuestaCompradorMail extends Mailable
    {
        use Queueable, SerializesModels;

        public function __construct(
            public AgendaPropuesta $propuesta,
            public string $urlAceptar,
            public string $urlRechazar,
        ) {}

        public function build()
        {
            $nombreComprador = $this->propuesta->comprador->nombre_empresa
                               ?? $this->propuesta->comprador->name;

            return $this
                ->subject('IMPULSATE: Propuesta de agenda — ' . $this->propuesta->evento->nombre)
                ->markdown('mails.agenda-propuesta-comprador')
                ->with([
                    'nombreComprador' => $nombreComprador,
                    'evento'          => $this->propuesta->evento,
                    'citas'           => $this->propuesta->citas->sortBy('slot_inicio'),
                    'urlAceptar'      => $this->urlAceptar,
                    'urlRechazar'     => $this->urlRechazar,
                ]);
        }
    }
