<?php
namespace App\Jobs;

use App\Mail\EncuestaRecordatorioMail;
use App\Models\EncuestaSatisfaccion;
use App\Models\User;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\InteractsWithQueue;
use Illuminate\Queue\SerializesModels;
use Illuminate\Support\Facades\Mail;

class SendRecordatorioJob implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public int $tries = 3;
    public int $timeout = 60;

    public function __construct(public int $encuestaId) {}

    public function handle(): void
    {
        $encuesta = EncuestaSatisfaccion::with(['evento', 'user', 'plantilla'])
            ->find($this->encuestaId);

        if (!$encuesta || $encuesta->completada_at) return;

        $email = $encuesta->user?->email ?? $encuesta->email_prueba;
        if (!$email) return;

        Mail::to($email)->send(new EncuestaRecordatorioMail($encuesta));
    }
}

─ 1D. Nuevo método en EncuestaAdminController ───────────────────────────────

    public function enviarRecordatorio(Request $request)
    {
        $request->validate([
            'evento_id' => 'required|exists:eventos,id',
        ]);

        // Buscar encuestas enviadas pero NO completadas
        $pendientes = EncuestaSatisfaccion::where('evento_id', $request->evento_id)
            ->where('es_prueba', false)
            ->whereNull('completada_at')
            ->get();

        if ($pendientes->isEmpty()) {
            return back()->with('info', 'Todos los participantes ya respondieron la encuesta. 🎉');
        }

        foreach ($pendientes as $encuesta) {
            \App\Jobs\SendRecordatorioJob::dispatch($encuesta->id)
                ->onQueue('encuestas');
        }

        return back()->with('success', "Recordatorio enviado a {$pendientes->count()} persona(s) que no han contestado.");
    }

─ 2C. Método reportePdf en EncuestaAdminController ──────────────────────────

    public function reportePdf(Request $request)
    {
        $eventoId    = $request->get('evento_id');
        $plantillaId = $request->get('plantilla_id');

        $plantilla = $plantillaId
            ? EncuestaPlantilla::find($plantillaId)
            : EncuestaPlantilla::where('activa', true)->first();

        $evento = $eventoId ? Evento::find($eventoId) : null;

        // Encuestas completadas
        $encuestas = EncuestaSatisfaccion::with(['user', 'respuestas', 'evento'])
            ->where('es_prueba', false)
            ->whereNotNull('completada_at')
            ->when($eventoId, fn($q) => $q->where('evento_id', $eventoId))
            ->orderBy('tipo')
            ->orderBy('completada_at')
            ->get();

        $totalEnviadas    = EncuestaSatisfaccion::where('es_prueba', false)
            ->when($eventoId, fn($q) => $q->where('evento_id', $eventoId))
            ->count();
        $totalRespondidas = $encuestas->count();
        $compradores      = $encuestas->where('tipo', 'comprador')->count();
        $proveedores      = $encuestas->where('tipo', 'proveedor')->count();

        // Agrupar respuestas por pregunta para gráficas generales
        $encuestaIds = $encuestas->pluck('id');

        $datosGenerales = EncuestaRespuesta::whereIn('encuesta_satisfaccion_id', $encuestaIds)
            ->whereNotIn('tipo', ['texto']) // excluir texto libre de gráficas
            ->selectRaw('pregunta, tipo, respuesta, COUNT(*) as total')
            ->groupBy('pregunta', 'tipo', 'respuesta')
            ->get()
            ->groupBy('pregunta')
            ->map(function($grupo) {
                $totalResp = $grupo->sum('total');
                return [
                    'tipo'     => $grupo->first()->tipo,
                    'total'    => $totalResp,
                    'opciones' => $grupo->mapWithKeys(fn($r) => [
                        $r->respuesta => [
                            'count'      => $r->total,
                            'porcentaje' => $totalResp > 0 ? round($r->total * 100 / $totalResp, 1) : 0,
                        ]
                    ]),
                ];
            });

        // Respuestas por persona
        $respuestasPorPersona = $encuestas->map(fn($enc) => [
            'nombre'   => $enc->user?->name ?? $enc->email_prueba ?? '—',
            'email'    => $enc->user?->email ?? '—',
            'tipo'     => $enc->tipo,
            'fecha'    => $enc->completada_at?->format('d/m/Y H:i'),
            'respuestas' => $enc->respuestas->map(fn($r) => [
                'pregunta'  => $r->pregunta,
                'respuesta' => $r->respuesta,
                'tipo'      => $r->tipo,
            ])->values(),
        ]);

        $pdf = \Barryvdh\DomPDF\Facade\Pdf::loadView('exports.encuesta-reporte', [
            'plantilla'           => $plantilla,
            'evento'              => $evento,
            'totalEnviadas'       => $totalEnviadas,
            'totalRespondidas'    => $totalRespondidas,
            'tasa'                => $totalEnviadas > 0 ? round($totalRespondidas * 100 / $totalEnviadas, 1) : 0,
            'compradores'         => $compradores,
            'proveedores'         => $proveedores,
            'datosGenerales'      => $datosGenerales,
            'respuestasPorPersona'=> $respuestasPorPersona,
            'fecha'               => now()->format('d/m/Y H:i'),
        ])->setPaper('a4', 'portrait');

        $filename = 'reporte-encuestas-' . now()->format('Y-m-d') . '.pdf';
        return $pdf->download($filename);
    }

    // Función para dibujar un canvas pastel
    function dibujarPastel(canvasId, datos) {
        const canvas = document.getElementById(canvasId)
        if (!canvas) return
        const ctx = canvas.getContext('2d')
        const opciones = Object.entries(datos.opciones)
        const total = opciones.reduce((s, [, v]) => s + v.count, 0)
        let startAngle = -Math.PI / 2
        ctx.clearRect(0, 0, canvas.width, canvas.height)
        const cx = canvas.width / 2
        const cy = canvas.height / 2
        const r  = Math.min(cx, cy) - 4
        opciones.forEach(([, stat], idx) => {
            const slice = (stat.count / total) * 2 * Math.PI
            ctx.beginPath()
            ctx.moveTo(cx, cy)
            ctx.arc(cx, cy, r, startAngle, startAngle + slice)
            ctx.closePath()
            ctx.fillStyle = COLORES_PASTEL[idx % COLORES_PASTEL.length]
            ctx.fill()
            ctx.strokeStyle = '#1f2937'
            ctx.lineWidth = 1
            ctx.stroke()
            startAngle += slice
        })
    }

    // Dibujar todos los pasteles al montar y cuando cambien datos
    import { watch, nextTick } from 'vue'
    watch([tipoPastel, vistaActiva], async () => {
        if (!tipoPastel.value || vistaActiva.value !== 'general') return
        await nextTick()
        if (props.plantilla) {

                if (d && ['opciones','multiple','binario'].includes(p.tipo)) {
                    dibujarPastel('pastel-' + idx, d)

─ 4D. Crear app/Exports/EncuestasResumenSheet.php ───────────────────────────

<?php

namespace App\Exports;

use App\Models\EncuestaPlantilla;
use App\Models\EncuestaRespuesta;
use App\Models\EncuestaSatisfaccion;
use Maatwebsite\Excel\Concerns\FromArray;
use Maatwebsite\Excel\Concerns\WithStyles;
use Maatwebsite\Excel\Concerns\WithTitle;
use Maatwebsite\Excel\Concerns\WithColumnWidths;
use PhpOffice\PhpSpreadsheet\Worksheet\Worksheet;
use PhpOffice\PhpSpreadsheet\Style\Alignment;
use PhpOffice\PhpSpreadsheet\Style\Fill;

class EncuestasResumenSheet implements FromArray, WithStyles, WithTitle, WithColumnWidths
{
    public function __construct(
        private ?int $eventoId,
        private ?EncuestaPlantilla $plantilla
    ) {}

    public function title(): string { return 'Resumen'; }

    public function columnWidths(): array
    {
        return ['A' => 55, 'B' => 35, 'C' => 12, 'D' => 12];
    }

    public function array(): array
    {
        $encuestaIds = EncuestaSatisfaccion::where('es_prueba', false)
            ->whereNotNull('completada_at')
            ->when($this->eventoId, fn($q) => $q->where('evento_id', $this->eventoId))
            ->pluck('id');

        $totalRespondidas   = $encuestaIds->count();
        $totalEnviadas      = EncuestaSatisfaccion::where('es_prueba', false)
            ->when($this->eventoId, fn($q) => $q->where('evento_id', $this->eventoId))
            ->count();

        $rows = [
            ['REPORTE DE ENCUESTAS DE SATISFACCIÓN', '', '', ''],
            ['Plantilla:', $this->plantilla?->nombre ?? '—', '', ''],
            ['Total enviadas:', $totalEnviadas, '', ''],
            ['Total respondidas:', $totalRespondidas, '', ''],
            ['Tasa de respuesta:', ($totalEnviadas > 0 ? round($totalRespondidas * 100 / $totalEnviadas, 1) : 0) . '%', '', ''],
            ['Generado:', now()->format('d/m/Y H:i'), '', ''],
            ['', '', '', ''],
            ['PREGUNTA', 'OPCIÓN DE RESPUESTA', 'CANTIDAD', '% DEL TOTAL'],
        ];

        $preguntas = $this->plantilla?->preguntas ?? [];

        foreach ($preguntas as $pregunta) {
            if ($pregunta['tipo'] === 'texto') {
                // Para texto libre, listar las respuestas
                $textos = EncuestaRespuesta::whereIn('encuesta_satisfaccion_id', $encuestaIds)
                    ->where('pregunta', $pregunta['texto'])
                    ->whereNotNull('respuesta')
                    ->where('respuesta', '!=', '')
                    ->pluck('respuesta');

                $rows[] = [$pregunta['texto'], '(Texto libre — ' . $textos->count() . ' respuestas)', '', ''];
                foreach ($textos as $txt) {
                    $rows[] = ['', '  • ' . $txt, '', ''];
                }
                $rows[] = ['', '', '', ''];
                continue;
            }

            $grupo = EncuestaRespuesta::whereIn('encuesta_satisfaccion_id', $encuestaIds)
                ->where('pregunta', $pregunta['texto'])
                ->selectRaw('respuesta, COUNT(*) as total')
                ->groupBy('respuesta')
                ->orderByDesc('total')
                ->get();

            $totalPregunta = $grupo->sum('total');
            $primerRow = true;

            foreach ($grupo as $g) {
                $pct = $totalPregunta > 0 ? round($g->total * 100 / $totalPregunta, 1) : 0;
                $rows[] = [
                    $primerRow ? $pregunta['texto'] : '',
                    $g->respuesta ?? '(sin respuesta)',
                    $g->total,
                    $pct . '%',
                ];
                $primerRow = false;
            }

            if ($grupo->isEmpty()) {
                $rows[] = [$pregunta['texto'], '(sin respuestas)', 0, '0%'];
            }

            $rows[] = ['', '', '', ''];
        }

        return $rows;
    }

    public function styles(Worksheet $sheet): array
    {
        // Título principal
        $sheet->mergeCells('A1:D1');
        $sheet->getStyle('A1')->applyFromArray([
            'font' => ['bold' => true, 'size' => 16, 'color' => ['rgb' => 'FFFFFF']],
            'fill' => ['fillType' => Fill::FILL_SOLID, 'startColor' => ['rgb' => '8B1028']],
            'alignment' => ['horizontal' => Alignment::HORIZONTAL_CENTER],
        ]);

        // Encabezados de tabla (fila 8)
        $sheet->getStyle('A8:D8')->applyFromArray([
            'font' => ['bold' => true, 'color' => ['rgb' => 'FFFFFF']],
            'fill' => ['fillType' => Fill::FILL_SOLID, 'startColor' => ['rgb' => '8B1028']],
        ]);

        return [];
    }
}

─ 4E. Crear app/Exports/EncuestasDetalleSheet.php ───────────────────────────

<?php

namespace App\Exports;

use App\Models\EncuestaPlantilla;
use App\Models\EncuestaSatisfaccion;
use Maatwebsite\Excel\Concerns\FromCollection;
use Maatwebsite\Excel\Concerns\WithHeadings;
use Maatwebsite\Excel\Concerns\WithStyles;
use Maatwebsite\Excel\Concerns\WithTitle;
use Maatwebsite\Excel\Concerns\WithColumnWidths;
use PhpOffice\PhpSpreadsheet\Worksheet\Worksheet;
use PhpOffice\PhpSpreadsheet\Style\Fill;

class EncuestasDetalleSheet implements FromCollection, WithHeadings, WithStyles, WithTitle, WithColumnWidths
{
    public function __construct(
        private ?int $eventoId,
        private ?EncuestaPlantilla $plantilla,
        private ?string $tipo  // null=todos, 'comprador', 'proveedor'
    ) {}

    public function title(): string
    {
        return match($this->tipo) {
            'comprador' => 'Compradores',
            'proveedor' => 'Proveedores',
            default     => 'Todas las Respuestas',
        };
    }

    public function columnWidths(): array
    {
        return ['A' => 25, 'B' => 30, 'C' => 14, 'D' => 20];
    }

    public function collection()
    {
        $preguntas = $this->plantilla?->preguntas ?? [];

        $query = EncuestaSatisfaccion::with(['evento', 'user', 'respuestas'])
            ->where('es_prueba', false)
            ->whereNotNull('completada_at')
            ->when($this->eventoId, fn($q) => $q->where('evento_id', $this->eventoId))
            ->when($this->tipo, fn($q) => $q->where('tipo', $this->tipo))
            ->orderBy('tipo')
            ->orderBy('completada_at');

        return $query->get()->map(function ($enc) use ($preguntas) {
            $respuestasMap = $enc->respuestas->pluck('respuesta', 'pregunta');

            $row = [
                'Participante'    => $enc->user?->name ?? '—',
                'Email'           => $enc->user?->email ?? '—',
                'Tipo'            => ucfirst($enc->tipo),
                'Fecha respuesta' => $enc->completada_at?->format('d/m/Y H:i'),
            ];

            foreach ($preguntas as $pregunta) {
                $row[$pregunta['texto']] = $respuestasMap[$pregunta['texto']] ?? '—';
            }

            return $row;
        });
    }

    public function headings(): array
    {
        $preguntas = $this->plantilla?->preguntas ?? [];
        $base = ['Participante', 'Email', 'Tipo', 'Fecha respuesta'];
        foreach ($preguntas as $p) {
            $base[] = $p['texto'];
        }
        return $base;
    }

    public function styles(Worksheet $sheet): array
    {
        return [
            1 => [
                'font' => ['bold' => true, 'color' => ['rgb' => 'FFFFFF']],
                'fill' => ['fillType' => Fill::FILL_SOLID, 'startColor' => ['rgb' => '8B1028']],
            ],
        ];
    }
}
