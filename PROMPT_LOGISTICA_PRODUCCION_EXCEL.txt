    public function calcularPerfilCompleto(): bool
    {
        $infoBasica = filled($this->nombre_restaurante)
            && filled($this->descripcion)
            && filled($this->telefono)
            && filled($this->municipio);

        $tieneLogo = filled($this->logo_path);

        $tieneProducto = !empty($this->productos_top)
            && collect($this->productos_top)->contains(
                fn($p) => filled($p['nombre'] ?? ($p ?? null))
            );

        $facturaOk = !$this->factura
            || (filled($this->rfc) && filled($this->regimen_fiscal));

        $creditoOk = !$this->acepta_credito
            || (
                filled($this->credito_monto_maximo)
                && filled($this->credito_tiempo_cantidad)
                && filled($this->credito_tiempo_unidad)
            );

        // Logística: el campo debe haber sido respondido (no null).
        // Si respondió que sí entrega, cobertura y forma son obligatorios.
        $logisticaOk = !is_null($this->entrega_domicilio)
            && (
                !$this->entrega_domicilio
                || (filled($this->cobertura_entrega) && filled($this->forma_entrega))
            );

        return $infoBasica && $tieneLogo && $tieneProducto
            && $facturaOk && $creditoOk && $logisticaOk;
    }

<?php

namespace App\Exports;

use App\Models\Restaurantero;
use Maatwebsite\Excel\Concerns\FromCollection;
use Maatwebsite\Excel\Concerns\WithHeadings;

use PhpOffice\PhpSpreadsheet\Style\Alignment;
use PhpOffice\PhpSpreadsheet\Style\Border;
use PhpOffice\PhpSpreadsheet\Style\Fill;

class ProveedoresExport implements FromCollection, WithHeadings, WithStyles, WithColumnWidths, WithEvents
{
    public function collection()
    {
        return Restaurantero::with('user')
            ->withCount('citas')
            ->orderBy('nombre_restaurante')
            ->get()
            ->map(function ($r) {

                // Productos: nombres separados por coma
                $productos = collect($r->productos_top ?? [])
                    ->filter(fn($p) => filled($p['nombre'] ?? null))
                    ->map(fn($p) => $p['nombre'])
                    ->join(', ') ?: '—';

                // Capacidades: "NombreProducto: X unidad/mes"
                $capacidades = collect($r->productos_top ?? [])
                    ->filter(fn($p) => isset($p['capacidad_cantidad']) && $p['capacidad_cantidad'] !== null)
                    ->map(fn($p) =>
                        ($p['nombre'] ?? '?') . ': '
                        . number_format((float)$p['capacidad_cantidad'], 0, '.', ',')
                        . ' ' . ($p['capacidad_unidad'] ?? '')
                        . '/mes'
                    )
                    ->join(' | ') ?: '—';

                $plazo = ($r->credito_tiempo_cantidad && $r->credito_tiempo_unidad)
                    ? "{$r->credito_tiempo_cantidad} {$r->credito_tiempo_unidad}"
                    : '—';

                $bool = fn($val) => is_null($val) ? '—' : ($val ? 'Sí' : 'No');

                return [
                    // ── Identificación ──────────────────────────────────
                    'Empresa'               => $r->nombre_restaurante ?? '—',
                    'Categoría'             => $r->categoria ?? '—',
                    'Razón Social'          => $r->razon_social ?? '—',
                    'Representante'         => $r->nombre_representante ?? '—',
                    'CURP'                  => $r->curp_representante ?? '—',
                    // ── Ubicación y contacto ─────────────────────────────
                    'Municipio'             => $r->municipio ?? '—',
                    'Dom. en Yucatán'       => $bool($r->domicilio_en_yucatan),
                    'Teléfono'              => $r->telefono ?? '—',
                    'Email'                 => $r->user?->email ?? '—',
                    'Sitio Web'             => $r->sitio_web ?? '—',
                    // ── Negocio ──────────────────────────────────────────
                    'RFC'                   => $r->rfc ?? '—',
                    'Núm. Empleados'        => $r->num_empleados ?? '—',
                    'Inicio Operaciones'    => $r->fecha_inicio_operaciones ?? '—',
                    'Mercado Meta'          => $r->mercado_meta ?? '—',
                    'Vida Anaquel'          => $r->tiempo_vida_anaquel ?? '—',
                    'Req. Refrigeración'    => $bool($r->requiere_refrigeracion),
                    'Req. Congelación'      => $bool($r->requiere_congelacion),
                    // ── Productos ────────────────────────────────────────
                    'Productos / Servicios' => $productos,
                    'Capacidades Producción'=> $capacidades,
                    // ── Logística ────────────────────────────────────────
                    'Entrega Domicilio'     => $bool($r->entrega_domicilio),
                    'Cobertura'             => $r->cobertura_entrega ? ucfirst($r->cobertura_entrega) : '—',
                    'Forma Entrega'         => $r->forma_entrega ? ucfirst($r->forma_entrega) : '—',
                    // ── Condiciones comerciales ───────────────────────────
                    'Acepta Crédito'        => $bool($r->acepta_credito),
                    'Crédito Máx. (MXN)'   => $r->credito_monto_maximo
                                                ? number_format((float)$r->credito_monto_maximo, 2, '.', ',')
                                                : '—',
                    'Plazo Crédito'         => $plazo,
                    'Contraentrega'         => $bool($r->pago_contraentrega),
                    'Factura'               => $bool($r->factura),
                    'Régimen Fiscal'        => $r->regimen_fiscal ?? '—',
                    // ── Plataforma ───────────────────────────────────────
                    'Total Citas'           => $r->citas_count,
                    'Aprobado'              => $bool($r->aprobado),
                    'Activo'                => $bool($r->activo),
                    'Perfil Completo'       => $bool($r->perfil_completo),
                    'Sol. Aprobación'       => $r->solicitado_aprobacion_at?->format('d/m/Y H:i') ?? '—',
                    'Registro'              => $r->created_at->format('d/m/Y'),
                ];
            });
    }

    public function styles(Worksheet $sheet): array
    {
        $lastRow = $sheet->getHighestRow();
        $lastCol = $sheet->getHighestColumn();

        // Filas alternas: blanco y rosa-guinda muy suave
        for ($row = 2; $row <= $lastRow; $row++) {
            if ($row % 2 === 0) {
                $sheet->getStyle("A{$row}:{$lastCol}{$row}")->applyFromArray([
                    'fill' => [
                        'fillType'   => Fill::FILL_SOLID,
                        'startColor' => ['rgb' => 'FDF2F4'],
                    ],
                ]);
            }
        }

        // Bordes finos en toda la tabla
        $sheet->getStyle("A1:{$lastCol}{$lastRow}")->applyFromArray([
            'borders' => [
                'allBorders' => [
                    'borderStyle' => Border::BORDER_THIN,
                    'color'       => ['rgb' => 'E5E7EB'],
                ],
            ],
        ]);

        // Borde inferior grueso debajo del encabezado
        $sheet->getStyle("A1:{$lastCol}1")->applyFromArray([
            'borders' => [
                'bottom' => [
                    'borderStyle' => Border::BORDER_MEDIUM,
                    'color'       => ['rgb' => '6B0D1E'],
                ],
            ],
        ]);

        return [
            // Encabezado
            1 => [
                'font' => [
                    'bold'  => true,
                    'color' => ['rgb' => 'FFFFFF'],
                    'size'  => 10,
                    'name'  => 'Calibri',
                ],
                'fill' => [
                    'fillType'   => Fill::FILL_SOLID,
                    'startColor' => ['rgb' => '8B1028'],
                ],
                'alignment' => [
                    'vertical'   => Alignment::VERTICAL_CENTER,
                    'horizontal' => Alignment::HORIZONTAL_CENTER,
                    'wrapText'   => true,
                ],
            ],
            // Filas de datos
            "A2:{$lastCol}{$lastRow}" => [
                'font' => ['size' => 9, 'name' => 'Calibri'],
                'alignment' => ['vertical' => Alignment::VERTICAL_CENTER],
            ],
            // Columnas numéricas centradas (Total Citas, Aprobado, Activo, Perfil, etc.)
            "AC2:AH{$lastRow}" => [
                'alignment' => ['horizontal' => Alignment::HORIZONTAL_CENTER],
            ],
            // Columna Empresa en negrita
            "A2:A{$lastRow}" => [
                'font' => ['bold' => true, 'size' => 9],
            ],
        ];
    }

    public function registerEvents(): array
    {
        return [
            AfterSheet::class => function (AfterSheet $event) {
                $ws      = $event->sheet->getDelegate();
                $lastCol = $ws->getHighestColumn();
                $lastRow = $ws->getHighestRow();

                // Altura del encabezado
                $ws->getRowDimension(1)->setRowHeight(32);

                // Altura uniforme de filas de datos
                for ($r = 2; $r <= $lastRow; $r++) {
                    $ws->getRowDimension($r)->setRowHeight(18);
                }

                // Congelar encabezado
                $ws->freezePane('A2');

                // Autofiltro
                $ws->setAutoFilter("A1:{$lastCol}1");

                // Nombre de pestaña y color
                $ws->setTitle('Proveedores');
                $ws->getTabColor()->setARGB('FF8B1028');
            },
        ];
    }

    npm run build

PASO B — Subir archivos al servidor de producción via FTP/Git/rsync:
    Los archivos modificados en este prompt son:
    • app/Models/Restaurantero.php
    • app/Http/Controllers/ProveedorPerfilController.php
    • app/Exports/ProveedoresExport.php
    • resources/js/Pages/Restaurantero/Panel.vue

    • public/build/ (assets compilados)

PASO C — Ejecutar en el servidor de producción (via SSH o panel de hosting):

    # 1. Correr la migración (solo la nueva — no toca datos existentes)
    php artisan migrate --force

    # 2. Limpiar y recompilar caché de configuración y rutas
    php artisan config:clear
    php artisan config:cache
    php artisan route:clear
    php artisan route:cache

    # 3. Limpiar vistas compiladas
    php artisan view:clear

    # 4. Optimizar autoloader
    php artisan optimize

    # 5. Si el proyecto usa colas (queues), reiniciarlas
    php artisan queue:restart

NOTAS IMPORTANTES DEL DEPLOY:
• La migración solo AGREGA columnas nullable — no rompe datos existentes.
• Proveedores ya registrados tendrán entrega_domicilio = NULL (no afecta perfil_completo
  actual, pero tras este deploy sí lo afectará → los proveedores existentes necesitarán
  llenar la sección de logística para recalcular perfil_completo = true).
• Si deseas que los proveedores existentes no pierdan perfil_completo, agrega un seeder
  opcional que marque entrega_domicilio = false en quienes ya tengan perfil_completo = true.
  Comando de ejemplo:
      Restaurantero::where('perfil_completo', true)
          ->whereNull('entrega_domicilio')
          ->update(['entrega_domicilio' => false]);
  Puedes correr esto como un tinker one-liner antes o después de migrate:
      php artisan tinker --execute="App\Models\Restaurantero::where('perfil_completo',true)->whereNull('entrega_domicilio')->update(['entrega_domicilio'=>false]);"
