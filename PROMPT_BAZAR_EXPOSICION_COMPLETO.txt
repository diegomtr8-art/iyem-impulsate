    private function sincronizarCriterios(Evento $evento, Request $request): void
    {
        $evento->criterios()->delete();

        if ($request->boolean('con_criterios_evaluacion') && !empty($request->criterios)) {
            foreach ($request->criterios as $index => $criterio) {
                EventoCriterio::create([
                    'evento_id'  => $evento->id,
                    'nombre'     => $criterio['nombre'],
                    'porcentaje' => $criterio['porcentaje'],
                    'orden'      => $index,
                ]);
            }
        }
    }

    // Llamar al final de store() y update():

─ 4A. index($eventoId) — Lista de participantes con evaluaciones ────────────

    public function index(Evento $evento)
    {
        // Validar que sea bazar con criterios
        abort_if(!$evento->esBazar(), 404, 'Este evento no es de tipo Bazar/Exposición.');

        $criterios = $evento->criterios()->get();

        $participantes = DB::table('evento_usuario as eu')
            ->join('users as u', 'u.id', '=', 'eu.user_id')
            ->where('eu.evento_id', $evento->id)
            ->where('eu.estado', 'aprobado')
            ->select([
                'eu.user_id', 'u.name', 'u.email',
                'eu.puntaje_total', 'eu.seleccionado',
                'eu.correo_aprobacion_enviado', 'eu.correo_rechazo_enviado',
            ])
            ->orderByDesc('eu.puntaje_total')
            ->get()
            ->map(function ($p) use ($criterios, $evento) {
                // Cargar puntajes por criterio
                $evaluaciones = EventoEvaluacion::where('evento_id', $evento->id)
                    ->where('user_id', $p->user_id)
                    ->pluck('puntaje', 'criterio_id');

                $p->evaluaciones = $criterios->map(fn($c) => [
                    'criterio_id' => $c->id,
                    'nombre'      => $c->nombre,
                    'porcentaje'  => $c->porcentaje,
                    'puntaje'     => $evaluaciones[$c->id] ?? null,
                ]);
                $p->evaluado = $evaluaciones->count() === $criterios->count();
                return $p;
            });

        $totalSolicitudes = DB::table('evento_usuario')
            ->where('evento_id', $evento->id)->count();

        $seleccionados = DB::table('evento_usuario')
            ->where('evento_id', $evento->id)
            ->where('seleccionado', true)->count();

        return Inertia::render('Admin/Eventos/BazarEvaluacion', [
            'evento'      => $evento->only('id', 'nombre', 'max_espacios', 'con_criterios_evaluacion'),
            'criterios'   => $criterios,
            'participantes' => $participantes,
            'kpis' => [
                'total_solicitudes' => $totalSolicitudes,
                'aprobados'         => $participantes->count(),
                'evaluados'         => $participantes->where('evaluado', true)->count(),
                'seleccionados'     => $seleccionados,
                'max_espacios'      => $evento->max_espacios,
                'disponibles'       => max(0, ($evento->max_espacios ?? 0) - $seleccionados),
            ],
        ]);
    }

─ 4B. evaluar($evento, $userId) — Guardar puntajes por criterio ────────────

    public function evaluar(Request $request, Evento $evento, int $userId)
    {
        $request->validate([
            'evaluaciones'               => 'required|array',
            'evaluaciones.*.criterio_id' => 'required|exists:evento_criterios,id',
            'evaluaciones.*.puntaje'     => 'required|numeric|min:0|max:100',
        ]);

        $puntajeTotal = 0;

        foreach ($request->evaluaciones as $eval) {
            $criterio = EventoCriterio::find($eval['criterio_id']);

            EventoEvaluacion::updateOrCreate(
                ['evento_id' => $evento->id, 'user_id' => $userId, 'criterio_id' => $eval['criterio_id']],
                ['puntaje'   => $eval['puntaje']]
            );

            // Puntaje ponderado: (puntaje/100) * porcentaje
            $puntajeTotal += ($eval['puntaje'] / 100) * $criterio->porcentaje;
        }

        // Guardar puntaje total en evento_usuario
        DB::table('evento_usuario')
            ->where('evento_id', $evento->id)
            ->where('user_id', $userId)
            ->update(['puntaje_total' => round($puntajeTotal, 2)]);

        return back()->with('success', 'Evaluación guardada.');
    }

─ 4C. toggleSeleccion($evento) — Seleccionar/deseleccionar participante ────

    public function toggleSeleccion(Request $request, Evento $evento)
    {
        $request->validate([
            'user_id'     => 'required|integer|exists:users,id',
            'seleccionado' => 'required|boolean',
        ]);

        // Si está intentando seleccionar, verificar que no exceda max_espacios
        if ($request->seleccionado && $evento->max_espacios) {
            $actuales = DB::table('evento_usuario')
                ->where('evento_id', $evento->id)
                ->where('seleccionado', true)
                ->count();

            if ($actuales >= $evento->max_espacios) {
                return back()->withErrors([
                    'error' => "Ya alcanzaste el límite de {$evento->max_espacios} espacios."
                ]);
            }
        }

        DB::table('evento_usuario')
            ->where('evento_id', $evento->id)
            ->where('user_id', $request->user_id)
            ->update(['seleccionado' => $request->seleccionado]);

        return back()->with('success', $request->seleccionado
            ? 'Participante seleccionado.'
            : 'Participante deseleccionado.');
    }

─ 4D. enviarAprobacion($evento) — Solo a los seleccionados ────────────────

    public function enviarAprobacion(Request $request, Evento $evento)
    {
        $seleccionados = DB::table('evento_usuario as eu')
            ->join('users as u', 'u.id', '=', 'eu.user_id')
            ->where('eu.evento_id', $evento->id)
            ->where('eu.seleccionado', true)
            ->where('eu.correo_aprobacion_enviado', false)
            ->select('eu.user_id', 'u.name', 'u.email')
            ->get();

        if ($seleccionados->isEmpty()) {
            return back()->withErrors(['error' => 'No hay participantes seleccionados pendientes de notificar.']);
        }

        $plantilla = \App\Models\PlantillaCorreo::paraClave('bazar_seleccionado');
        if (!$plantilla) {
            return back()->withErrors(['error' => 'Plantilla "bazar_seleccionado" no encontrada.']);
        }

        $enviados = 0;
        foreach ($seleccionados as $p) {
            try {
                $contenido = $plantilla->reemplazar([
                    'nombre_usuario' => $p->name,
                    'nombre_evento'  => $evento->nombre,
                ]);

                \Illuminate\Support\Facades\Mail::to($p->email)
                    ->queue(new \App\Mail\PlantillaCorreoMail(
                        (object) ['asunto' => $plantilla->asunto, 'contenido' => $contenido],
                        ['nombre_usuario' => $p->name]
                    ));

                DB::table('evento_usuario')
                    ->where('evento_id', $evento->id)
                    ->where('user_id', $p->user_id)
                    ->update(['correo_aprobacion_enviado' => true]);

                $enviados++;
            } catch (\Exception $e) {
                \Log::warning("Error correo aprobación bazar user {$p->user_id}: " . $e->getMessage());
            }
        }

        return back()->with('success', "Correo de aprobación enviado a {$enviados} participante(s).");
    }

    public function enviarRechazo(Request $request, Evento $evento)
    {
        // No seleccionados = aprobados pero no están en los N espacios
        $rechazados = DB::table('evento_usuario as eu')
            ->join('users as u', 'u.id', '=', 'eu.user_id')
            ->where('eu.evento_id', $evento->id)
            ->where('eu.estado', 'aprobado')
            ->where('eu.seleccionado', false)
            ->where('eu.correo_rechazo_enviado', false)
            ->select('eu.user_id', 'u.name', 'u.email', 'eu.token_evaluacion')
            ->get();

        if ($rechazados->isEmpty()) {
            return back()->withErrors(['error' => 'No hay participantes pendientes de notificar rechazo.']);
        }

        $plantilla = \App\Models\PlantillaCorreo::paraClave('bazar_rechazado');
        if (!$plantilla) {
            return back()->withErrors(['error' => 'Plantilla "bazar_rechazado" no encontrada.']);
        }

        $enviados = 0;
        foreach ($rechazados as $p) {
            try {
                // Generar token único si no tiene
                $token = $p->token_evaluacion ?? Str::random(48);

                if (!$p->token_evaluacion) {
                    DB::table('evento_usuario')
                        ->where('evento_id', $evento->id)
                        ->where('user_id', $p->user_id)
                        ->update(['token_evaluacion' => $token]);
                }

                $urlEvaluacion = route('bazar.evaluacion.publica', ['token' => $token]);

                $contenido = $plantilla->reemplazar([
                    'nombre_usuario'  => $p->name,
                    'nombre_evento'   => $evento->nombre,
                    'url_evaluacion'  => $urlEvaluacion,
                ]);

                \Illuminate\Support\Facades\Mail::to($p->email)
                    ->queue(new \App\Mail\PlantillaCorreoMail(
                        (object) ['asunto' => $plantilla->asunto, 'contenido' => $contenido],
                        ['nombre_usuario' => $p->name]
                    ));

                DB::table('evento_usuario')
                    ->where('evento_id', $evento->id)
                    ->where('user_id', $p->user_id)
                    ->update(['correo_rechazo_enviado' => true]);

                $enviados++;
            } catch (\Exception $e) {
                \Log::warning("Error correo rechazo bazar user {$p->user_id}: " . $e->getMessage());
            }
        }

        return back()->with('success', "Correo de rechazo enviado a {$enviados} participante(s).");
    }

    public function evaluacion(string $token)
    {
        $registro = DB::table('evento_usuario as eu')
            ->join('users as u', 'u.id', '=', 'eu.user_id')
            ->join('eventos as e', 'e.id', '=', 'eu.evento_id')
            ->where('eu.token_evaluacion', $token)
            ->select([
                'u.name', 'e.nombre as nombre_evento', 'e.id as evento_id',
                'eu.puntaje_total', 'eu.user_id',
            ])
            ->first();

        abort_if(!$registro, 404, 'Enlace de evaluación no encontrado o inválido.');

        // Cargar criterios y puntajes del participante
        $criterios = DB::table('evento_criterios as c')
            ->leftJoin('evento_evaluaciones as e', function ($j) use ($registro) {
                $j->on('e.criterio_id', '=', 'c.id')
                  ->where('e.user_id', $registro->user_id)
                  ->where('e.evento_id', $registro->evento_id);
            })
            ->where('c.evento_id', $registro->evento_id)
            ->orderBy('c.orden')
            ->select([
                'c.nombre', 'c.porcentaje',
                'e.puntaje',
            ])
            ->get()
            ->map(fn($c) => [
                'nombre'          => $c->nombre,
                'porcentaje'      => $c->porcentaje,
                'puntaje'         => $c->puntaje ?? 0,
                'puntaje_maximo'  => 100,
                'puntaje_ponderado' => round(($c->puntaje ?? 0) / 100 * $c->porcentaje, 2),
            ]);

        return Inertia::render('Bazar/EvaluacionPublica', [
            'nombre_usuario'  => $registro->name,
            'nombre_evento'   => $registro->nombre_evento,
            'puntaje_total'   => $registro->puntaje_total,
            'criterios'       => $criterios,
        ]);
    }
