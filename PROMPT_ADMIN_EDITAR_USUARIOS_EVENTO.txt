Archivo: app/Http/Controllers/AdminProveedorController.php

    <?php

    namespace App\Http\Controllers;

    use App\Models\Restaurantero;
    use App\Models\Categoria;
    use Illuminate\Http\Request;
    use Inertia\Inertia;

    class AdminProveedorController extends Controller
    {
        public function editar(Restaurantero $restaurantero)
        {
            $restaurantero->load('user:id,name,email', 'categoria:id,nombre');

            return Inertia::render('Admin/Proveedores/Editar', [
                'restaurantero' => $restaurantero,
                'categorias'    => Categoria::orderBy('nombre')->get(['id', 'nombre']),
            ]);
        }

        public function actualizar(Request $request, Restaurantero $restaurantero)
        {
            $validated = $request->validate([
                'nombre'              => 'required|string|max:255',
                'descripcion'         => 'nullable|string|max:2000',
                'telefono'            => 'nullable|string|max:30',
                'municipio'           => 'nullable|string|max:100',
                'rfc'                 => 'nullable|string|max:13',
                'sitio_web'           => 'nullable|url|max:255',
                'categoria_id'        => 'nullable|exists:categorias,id',
                'credito_tipo'        => 'nullable|string|max:100',
                'pago_contraentrega'  => 'nullable|boolean',
                'acepta_factura'      => 'nullable|boolean',
                'entrega_domicilio'   => 'nullable|boolean',
                'cobertura'           => 'nullable|string|max:255',
                'forma_entrega'       => 'nullable|string|max:255',
                // nombre y email del User relacionado
                'user_name'           => 'required|string|max:255',
                'user_email'          => 'required|email|max:255|unique:users,email,' . $restaurantero->user_id,
            ]);

            // Actualizar datos del restaurantero
            $restaurantero->update(collect($validated)
                ->except(['user_name', 'user_email'])
                ->toArray());

            // Actualizar datos del User
            $restaurantero->user->update([
                'name'  => $validated['user_name'],
                'email' => $validated['user_email'],
            ]);

            return redirect()->route('admin.proveedores.editar', $restaurantero)
                ->with('success', 'Proveedor actualizado correctamente.');
        }
    }

            ]);

        public function actualizar(Request $request, User $user)
        {
            abort_unless($user->hasRole('cliente'), 404);

    public function agregarProveedor(Request $request, \App\Models\Evento $evento)
    {
        $request->validate([
            'restaurantero_id' => 'required|exists:restauranteros,id',
        ]);

        $restaurantero = \App\Models\Restaurantero::findOrFail($request->restaurantero_id);

        // Verificar que no esté ya registrado como proveedor en este evento
        $yaRegistrado = $evento->usuarios()
            ->where('user_id', $restaurantero->user_id)
            ->where('tipo', 'proveedor')
            ->exists();

        if ($yaRegistrado) {
            return back()->with('error', 'Este proveedor ya está registrado en el evento.');
        }

        $evento->usuarios()->attach($restaurantero->user_id, [
            'tipo'          => 'proveedor',
            'estado'        => 'aprobado',
            'respondido_at' => now(),
        ]);

        return back()->with('success',
            'Proveedor "' . $restaurantero->nombre . '" agregado al evento correctamente.');
    }

    public function agregarComprador(Request $request, \App\Models\Evento $evento)
    {
        $request->validate([
            'user_id' => 'required|exists:users,id',
        ]);

        $user = \App\Models\User::role('cliente')->findOrFail($request->user_id);

        // Verificar que no esté ya registrado como comprador en este evento
        $yaRegistrado = $evento->usuarios()
            ->where('user_id', $user->id)
            ->where('tipo', 'comprador')
            ->exists();

        if ($yaRegistrado) {
            return back()->with('error', 'Este comprador ya está registrado en el evento.');
        }

        $evento->usuarios()->attach($user->id, [
            'tipo'          => 'comprador',
            'estado'        => 'aprobado',
            'respondido_at' => now(),
        ]);

        return back()->with('success',
            'Comprador "' . ($user->nombre_empresa ?? $user->name) . '" agregado al evento correctamente.');
    }

    public function gestionar(\App\Models\Evento $evento)
    {
        $evento->load([
            'usuarios' => fn($q) => $q->withPivot('tipo', 'estado', 'respondido_at')
        ]);

        // Separar proveedores y compradores ya registrados
        $proveedoresEnEvento = $evento->usuarios
            ->where('pivot.tipo', 'proveedor')
            ->values();

        $compradoresEnEvento = $evento->usuarios
            ->where('pivot.tipo', 'comprador')
            ->values();

        return Inertia::render('Admin/Eventos/Gestionar', [
            'evento'              => $evento,
            'proveedoresEnEvento' => $proveedoresEnEvento,
            'compradoresEnEvento' => $compradoresEnEvento,
            'todosProveedores'    => \App\Models\Restaurantero::orderBy('nombre')
                                        ->get(['id', 'nombre']),
            'todosCompradores'    => \App\Models\User::role('cliente')
                                        ->orderBy('name')

1. EDITAR PROVEEDOR:

   d) Guarda — debe mostrar "✅ Proveedor actualizado correctamente."
   e) Verifica que el cambio se refleja en la lista y en el perfil público

2. EDITAR COMPRADOR:
   a) Ve a Admin → Compradores (lista)
   b) Haz clic en "Editar" en cualquier comprador
   c) Modifica nombre, email
   d) Guarda — debe mostrar "✅ Comprador actualizado correctamente."

3. AGREGAR PROVEEDOR AL EVENTO:
   a) Ve a Admin → Eventos → Gestionar participantes
   b) En "Agregar proveedor", selecciona uno de la lista
   c) Haz clic en Agregar
   d) Verifica en BD: SELECT * FROM evento_usuario WHERE tipo='proveedor' AND estado='aprobado'
   e) El proveedor debe aparecer ahora en la lista de aprobados del evento

4. AGREGAR COMPRADOR AL EVENTO:
   a) En la misma vista, agrega un comprador manualmente
   b) Verifica en BD que se creó el registro con tipo='comprador', estado='aprobado'
   c) El comprador debe poder recibir propuestas de agenda

5. DUPLICADOS:
   a) Intenta agregar el mismo proveedor/comprador dos veces
   b) Debe mostrar: "Este proveedor/comprador ya está registrado en el evento."
